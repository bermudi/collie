# scripts/strip-manifest.txt — every upstream path Pup deliberately does not carry (ADR 9004).
#
# `scripts/check-strip.ts` reads this file and fails the merge when an upstream file is absent
# from Pup WITHOUT a line here (a silent casualty — the rename/delete drops git resolves without
# conflict), or when a file matching the strip patterns IS tracked (the strip leaked back in).
#
# Line shapes (fnmatch, `*` matches within one path segment only):
#   exact/path.ts        that file
#   dir/prefix/          everything under the directory
#   name-*.ts            glob
# Keep each entry commented with the reason. Adding a line is a decision: name the ADR.
